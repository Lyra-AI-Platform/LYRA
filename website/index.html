async function verifyHuman(token) { const res = await fetch('https://lyraauth.com/api/auth/siteverify', { method: 'POST', body: new URLSearchParams({ secret: 'YOUR_SECRET_KEY', token }) }); return (await res.json()).success; }
function verifyHuman($token) { $r = file_get_contents('https://lyraauth.com/api/auth/siteverify', false, stream_context_create(['http' => [ 'method' => 'POST', 'content' => http_build_query([ 'secret' => 'YOUR_SECRET_KEY', 'token' => $token ]) ]]) ); return json_decode($r)->success ?? false; }

Tremor Index · experimental

Read a stock from public data,
with every source shown

The Tremor Index reads a US stock from its price trend, the VIX, on-topic news headlines, SEC insider and fund filings and politicians' disclosures. Each reading shows which sources responded and links to the filings behind it.

It is general information, not financial advice. Our backtest over 4 years and 99 large US stocks found the price-and-VIX model does not yet beat simple baselines, so readings show a lean, not exact odds. Every reading's outcome is published.
